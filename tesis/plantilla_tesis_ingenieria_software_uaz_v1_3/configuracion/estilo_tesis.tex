% Fecha institucional automática. La fecha se toma del sistema que compila el
% documento (por ejemplo, los servidores de Overleaf), por lo que no requiere
% consultar servicios externos ni escribir el mes o el año manualmente.
\newcommand{\nombreMesActual}{%
  \ifcase\month
  \or enero\or febrero\or marzo\or abril\or mayo\or junio
  \or julio\or agosto\or septiembre\or octubre\or noviembre\or diciembre
  \fi
}
\newcommand{\mesAnioAutomatico}{\nombreMesActual\ de \the\year}
\newcommand{\fechaCompletaAutomatica}{\the\day\ de \nombreMesActual\ de \the\year}
